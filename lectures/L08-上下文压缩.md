## 详细讲解

### 1. 压缩触发公式

```typescript
const COMPACT_CONFIG = {
  contextWindow: 128000,    // 上下文窗口大小
  reserveTokens: 16384,     // 保留给回复的 token 数
  keepRecentTokens: 20000,  // 保留最近消息的 token 数
};

function shouldCompact(currentTokens: number): boolean {
  const threshold = COMPACT_CONFIG.contextWindow - COMPACT_CONFIG.reserveTokens;
  return currentTokens > threshold;
}
```

### 2. 切割点算法

```typescript
function findCutPoint(
  entries: SessionEntry[],
  keepRecentTokens: number
): { cutIndex: number; cutEntryId: string } | null {
  let tokensFromEnd = 0;
  let cutIndex = -1;

  // 从最新消息往回累计 token
  for (let i = entries.length - 1; i >= 0; i--) {
    tokensFromEnd += entries[i].tokenCount;

    if (tokensFromEnd >= keepRecentTokens) {
      cutIndex = i;
      break;
    }
  }

  if (cutIndex === -1) return null;

  // 找到合法切点（只能是 user / assistant，不能是 toolResult）
  for (let i = cutIndex; i >= 0; i--) {
    const entry = entries[i];
    if (entry.type === "user" || entry.type === "assistant") {
      // 确保不是 toolResult 的 toolCall 部分
      if (entry.type === "assistant" && entry.stopReason === "toolUse") {
        continue;
      }
      return { cutIndex: i, cutEntryId: entry.id };
    }
  }

  return null;
}
```

### 3. toolResult 不切割约束

```
正确切割点:  user / assistant (stopReason="stop")
错误切割点:  toolResult / assistant (stopReason="toolUse")

原因: toolResult 必须紧跟对应的 toolCall，不能分离
```

### 4. CompactionEntry 结构

```typescript
interface CompactionEntry extends BaseEntry {
  type: "compaction";
  summary: string;              // 压缩摘要
  firstKeptEntryId: string;     // 第一个保留的条目 ID

```

### 5. 迭代式压缩

```
压缩前:  entry1 → entry2 → ... → entryN (128k tokens)
                ↓
压缩后:  [摘要] → entryK → ... → entryN (40k tokens)
                ↓
再次压缩: [新摘要] → entryM → ... → entryN (30k tokens)
```

## 例题

**题目**：实现一个简化版的上下文压缩函数，当消息超过阈值时，将早期消息合并为摘要。

**参考答案**：

```typescript
function compactMessages(
  messages: Message[],
  maxTokens: number
): { messages: Message[]; compacted: boolean } {
  const totalTokens = messages.reduce((sum, m) => sum + estimateTokens(m), 0);

  if (totalTokens <= maxTokens) {
    return { messages, compacted: false };
  }

  // 找到需要保留的消息数量
  let tokensFromEnd = 0;
  let keepFromIndex = messages.length;

  for (let i = messages.length - 1; i >= 0; i--) {
    tokensFromEnd += estimateTokens(messages[i]);
    if (tokensFromEnd >= maxTokens * 0.6) {
      keepFromIndex = i;
      break;
    }
  }

  // 生成摘要
  const messagesToCompact = messages.slice(0, keepFromIndex);
  const summary = generateSummary(messagesToCompact);

  // 返回压缩后的消息
  return {
    messages: [
      { role: "system", content: summary },
      ...messages.slice(keepFromIndex),
    ],
    compacted: true,
  };
}

function estimateTokens(message: Message): number {
  // 简化的 token 估算
  return Math.ceil(JSON.stringify(message).length / 4);
}

function generateSummary(messages: Message[]): string {
  const userCount = messages.filter(m => m.role === "user").length;
  const assistantCount = messages.filter(m => m.role === "assistant").length;
  return `[压缩摘要] 包含 ${userCount} 个用户问题和 ${assistantCount} 个助手回复。`;
}
```
