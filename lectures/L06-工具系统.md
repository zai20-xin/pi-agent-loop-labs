function truncateHead(content: string, maxLines: number): { text: string; truncated: boolean } {
  const lines = content.split("\n");
  if (lines.length <= maxLines) {
    return { text: content, truncated: false };
  }
  return {
    text: lines.slice(0, maxLines).join("\n") + `\n\n[已截断，共 ${lines.length} 行]`,
    truncated: true,
  };
}

### 7. terminate 批量提前终止

`terminate` 是工具结果中的一个标志，用于指示是否应该在当前工具批次完成后提前终止 agent 循环。

```typescript
interface AgentToolResult<T> {
  // ...其他字段
  /**
   * Hint that the agent should stop after the current tool batch.
   * Early termination only happens when every finalized tool result 
   * in the batch sets this to true.
   */
  terminate?: boolean;
}
```

**终止规则**：只有当批次中**所有**工具的结果都设置了 `terminate: true` 时，才会触发提前终止。

```typescript
function shouldTerminateToolBatch(finalizedCalls: FinalizedToolCallOutcome[]): boolean {
  return finalizedCalls.length > 0 && finalizedCalls.every(
    (finalized) => finalized.result.terminate === true
  );
}
```

**使用场景**：
- 紧急停止：检测到严重错误时，所有工具同时返回终止信号
- 批次完成：所有工具任务完成后，不需要继续下一轮 LLM 调用
- 与 `beforeToolCall`/`afterToolCall` 配合：可在拦截器中设置终止标志

```typescript
// 示例：工具检测到致命错误时返回终止
return {
  content: [{ type: "text", text: "数据库连接失败" }],
  details: { error: "connection refused" },
  terminate: true,  // 请求终止
};
```

### 8. addedToolNames 动态引入新工具

`addedToolNames` 允许工具结果动态引入新的工具，使其从当前 transcript 点开始可用。

```typescript
interface AgentToolResult<T> {
  // ...其他字段
  /** Names of tools introduced by this result and available from this transcript point onward. */
  addedToolNames?: string[];
}
```

**实现原理**：在 `createToolResultMessage` 中，当工具结果包含 `addedToolNames` 时，会将其附加到 `ToolResultMessage` 中：

```typescript
function createToolResultMessage(finalized: FinalizedToolCallOutcome): ToolResultMessage {
  return {
    role: "toolResult",
    toolCallId: finalized.toolCall.id,
    toolName: finalized.toolCall.name,
    content: finalized.result.content ?? [],
    details: finalized.result.details,
    usage: finalized.result.usage,
    ...(finalized.result.addedToolNames?.length 
      ? { addedToolNames: finalized.result.addedToolNames } 
      : {}),
    isError: finalized.isError,
    timestamp: Date.now(),
  };
}
```

**使用场景**：
- 插件系统：工具执行后动态加载新插件
- 条件工具：根据运行时状态决定是否暴露某些工具
- 自适应接口：根据用户请求动态生成专用工具

```typescript
// 示例：执行后引入新的调试工具
return {
  content: [{ type: "text", text: "分析完成，发现性能瓶颈" }],
  details: { analysis: "..." },
  addedToolNames: ["profiler", "memory-analyzer"],  // 动态引入
};
```

### 9. executionMode 单工具级别覆盖

每个工具可以声明自己的执行模式，覆盖全局的 `toolExecution` 配置。

```typescript
interface AgentTool<TParameters extends TSchema, TDetails> {
  // ...其他字段
  /**
   * Per-tool execution mode override.
   * - "sequential": this tool must execute one at a time with other tool calls.
   * - "parallel": this tool can execute concurrently with other tool calls.
   *
   * If omitted, the default execution mode applies.
   */
  executionMode?: ToolExecutionMode;
}
```

**执行逻辑**：在 `executeToolCalls` 中，只要有一个工具声明了 `"sequential"`，整个批次就切换到顺序执行模式：

```typescript
async function executeToolCalls(
  currentContext: AgentContext,
  assistantMessage: AssistantMessage,
  config: AgentLoopConfig,
  signal: AbortSignal | undefined,
  emit: AgentEventSink,
): Promise<ExecutedToolCallBatch> {
  const toolCalls = assistantMessage.content.filter((c) => c.type === "toolCall");
  const hasSequentialToolCall = toolCalls.some(
    (tc) => currentContext.tools?.find((t) => t.name === tc.name)?.executionMode === "sequential",
  );
  if (config.toolExecution === "sequential" || hasSequentialToolCall) {
    return executeToolCallsSequential(currentContext, assistantMessage, toolCalls, config, signal, emit);
  }
  return executeToolCallsParallel(currentContext, assistantMessage, toolCalls, config, signal, emit);
}
```

**使用场景**：
- 数据库写入：需要避免并发冲突
- 资源竞争：工具访问共享资源时需要串行化
- 依赖关系：工具B的输入依赖工具A的输出

```typescript
// 示例：数据库写入工具声明顺序执行
const dbWriteTool: AgentTool<typeof WriteParams, WriteDetails> = {
  name: "dbWrite",
  label: "数据库写入",
  executionMode: "sequential",  // 强制顺序执行
  // ...
};
```

### 10. prepareArguments 参数兼容 shim

`prepareArguments` 是一个参数预处理函数，用于在 schema 校验前转换工具参数。

```typescript
interface AgentTool<TParameters extends TSchema, TDetails> {
  // ...其他字段
  /**
   * Optional compatibility shim for raw tool-call arguments before schema validation.
   * Must return an object that matches `TParameters`.
   */
  prepareArguments?: (args: unknown) => Static<TParameters>;
}
```

**处理流程**：在 `prepareToolCallArguments` 中应用转换：

```typescript
function prepareToolCallArguments(tool: AgentTool<any>, toolCall: AgentToolCall): AgentToolCall {
  if (!tool.prepareArguments) {
    return toolCall;
  }
  const preparedArguments = tool.prepareArguments(toolCall.arguments);
  if (preparedArguments === toolCall.arguments) {
    return toolCall;
  }
  return {
    ...toolCall,
    arguments: preparedArguments as Record<string, any>,
  };
}
```

**使用场景**：
- 向后兼容：适配不同版本的工具调用参数格式
- 参数标准化：统一参数命名约定（如 snake_case → camelCase）
- 默认值填充：为缺失的可选参数提供默认值
- 类型转换：将字符串参数转换为对应的类型

```typescript
// 示例：参数兼容 shim
const readFileTool: AgentTool<typeof ReadFileParams, ReadFileDetails> = {
  name: "readFile",
  label: "读取文件",
  parameters: ReadFileParams,

    return {

      encoding: (raw.encoding as string) || "utf-8",
      startLine: Number(raw.startLine || raw.start_line || 0),
    };
  },
  execute: async (toolCallId, params, signal?, onUpdate?) => {
    // params 已经是标准化后的格式
    // ...
  },
};
```

### 例题 3：beforeToolCall/afterToolCall 与 terminate 配合

**题目**：设计一个权限控制系统，在 `beforeToolCall` 中检查权限，并在检测到安全威胁时设置终止标志。

**参考答案**：

```typescript
import type { AgentLoopConfig, BeforeToolCallResult } from "../../pi/packages/agent/src/types.ts";

// 工具权限配置
const toolPermissions: Record<string, string[]> = {
  readFile: ["read"],
  writeFile: ["write"],
  deleteFile: ["delete", "admin"],
  databaseBackup: ["backup", "admin"],
};

// 权限检查函数
function hasPermission(toolName: string, userRoles: string[]): boolean {
  const requiredRoles = toolPermissions[toolName] || [];
  return requiredRoles.some(role => userRoles.includes(role));
}

// 安全威胁检测函数
function detectSecurityThreat(toolName: string, args: unknown): boolean {
  const params = args as Record<string, unknown>;
  
  // 检测路径遍历攻击
  if (params.path && typeof params.path === "string") {
    if (params.path.includes("..") || params.path.startsWith("/etc")) {
      return true;
    }
  }
  
  // 检测 SQL 注入
  if (params.query && typeof params.query === "string") {
    const sqlKeywords = ["DROP", "DELETE", "TRUNCATE", "ALTER"];
    if (sqlKeywords.some(keyword => params.query.toUpperCase().includes(keyword))) {
      return true;
    }
  }
  
  return false;
}

// 严重错误判断
function isCriticalError(details: unknown): boolean {
  if (!details || typeof details !== "object") return false;
  const d = details as Record<string, unknown>;
  return d.severity === "critical" || d.error === "EACCES";
}

// AgentLoopConfig 配置
const config: AgentLoopConfig = {
  // ...其他配置

  beforeToolCall: async (context, signal) => {
    const { toolCall, args, context: agentContext } = context;

    // 获取用户角色（示例）
    const userRoles = ["read", "write"];  // 实际中从 context 获取

    // 检查工具权限
    if (!hasPermission(toolCall.name, userRoles)) {
      return {
        block: true,
        reason: `无权执行 ${toolCall.name}，需要角色: ${toolPermissions[toolCall.name]?.join(", ")}`,
        terminate: true,  // 权限被拒绝时终止
      };
    }

    // 检测安全威胁
    if (detectSecurityThreat(toolCall.name, args)) {
      return {
        block: true,
        reason: "检测到安全威胁，操作已阻止",
        terminate: true,  // 安全威胁时终止
      };
    }

    return undefined;  // 允许执行
  },

  afterToolCall: async (context, signal) => {
    const { result, isError } = context;

    // 如果工具执行失败且错误严重，设置终止
    if (isError && isCriticalError(result.details)) {
      return {
        terminate: true,  // 严重错误时终止
      };
    }

    return undefined;
  },
};
```
