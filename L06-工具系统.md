### 5. 截断策略

```typescript
function truncateHead(content: string, maxLines: number): { text: string; truncated: boolean } {
  const lines = content.split("\n");
  if (lines.length <= maxLines) {
    return { text: content, truncated: false };
  }
  return {
    text: lines.slice(0, maxLines).join("\n") + `\n\n[已截断，共 ${lines.length} 行]`,
    truncated: true,
  };
}
```

## 例题

**题目**：实现一个 `searchFiles` 工具，在指定目录中搜索包含关键词的文件。

**要求**：
- 支持正则表达式搜索
- 支持限制最大结果数
- 返回匹配的文件路径和行号

**参考答案**：

```typescript
import { Type, type Static } from "../../pi/node_modules/typebox/build/index.mjs";
import type { AgentTool, AgentToolResult } from "../../pi/packages/agent/src/types.ts";

const SearchFilesParams = Type.Object({
  directory: Type.String({ description: "搜索目录" }),
  pattern: Type.String({ description: "搜索关键词（支持正则）" }),
  maxResults: Type.Optional(Type.Number({ description: "最大结果数", default: 100 })),
});

type SearchFilesParamsType = Static<typeof SearchFilesParams>;

interface SearchResult {
  filePath: string;
  lineNumber: number;
  line: string;
}

interface SearchFilesDetails {
  totalMatches: number;
  filesSearched: number;
  truncated: boolean;
}

const searchFilesTool: AgentTool<typeof SearchFilesParams, SearchFilesDetails> = {
  name: "searchFiles",
  label: "文件搜索",
  description: "在目录中搜索包含关键词的文件",
  parameters: SearchFilesParams,

  execute: async (toolCallId, params, signal?, onUpdate?) => {
    if (signal?.aborted) throw new Error("工具执行被中断");

    const results: SearchResult[] = [];
    const regex = new RegExp(params.pattern, "g");
    const maxResults = params.maxResults ?? 100;

    // 模拟搜索过程
    // 实际实现中会遍历目录、读取文件、匹配内容

    return {
      content: [
        {
          type: "text",
          text: `找到 ${results.length} 个匹配结果`,
        },
      ],
      details: {
        totalMatches: results.length,
        filesSearched: 10,
        truncated: results.length >= maxResults,
      },
    };
  },
};
```
